por isso a tabela `fundos` e alimentada em duas partes: classes (a maioria) e
fundos "standalone" que ainda nao tem nenhuma classe.

Uso:
    python pipeline/update_registro_fundo.py
"""

def baixar_e_extrair():
    config.RAW_DIR.mkdir(parents=True, exist_ok=True)
    destino_zip = config.RAW_DIR / "registro_fundo_classe.zip"

    print(f"Baixando registro fundo/classe de {config.CVM_REGISTRO_FUNDO_CLASSE_URL}")
    response = requests.get(config.CVM_REGISTRO_FUNDO_CLASSE_URL, stream=True, timeout=60)
    response.raise_for_status()
    with open(destino_zip, "wb") as arquivo:
        for chunk in response.iter_content(chunk_size=1024 * 1024):
            arquivo.write(chunk)

    with zipfile.ZipFile(destino_zip) as zf:
        zf.extract("registro_fundo.csv", config.RAW_DIR)
        zf.extract("registro_classe.csv", config.RAW_DIR)

    return config.RAW_DIR / "registro_fundo.csv", config.RAW_DIR / "registro_classe.csv"


def ler_csv(caminho):
    df = pd.read_csv(caminho, sep=config.CVM_CSV_SEPARATOR, encoding=config.CVM_CSV_ENCODING, dtype=str)
    print(f"{caminho.name} lido: {len(df)} linhas, {len(df.columns)} colunas")
    return df


def construir_classes(df_fundo, df_classe):
    """Classes (registro_classe.csv) herdando administrador/gestor do fundo pai."""
    cols_fundo = df_fundo[[
        "ID_Registro_Fundo", "CNPJ_Fundo", "Administrador", "CNPJ_Administrador",
        "Gestor", "CPF_CNPJ_Gestor", "Data_Inicio_Exercicio_Social", "Data_Fim_Exercicio_Social",
    ]]
    merged = df_classe.merge(cols_fundo, on="ID_Registro_Fundo", how="inner")

    out = pd.DataFrame()
    out["cnpj"] = merged["CNPJ_Classe"]
    out["fundo_cnpj_pai"] = merged["CNPJ_Fundo"]
    out["id_registro_fundo"] = merged["ID_Registro_Fundo"]
    out["id_registro_classe"] = merged["ID_Registro_Classe"]
    out["cd_cvm"] = merged["Codigo_CVM"]
    out["denominacao_social"] = merged["Denominacao_Social"]
    out["tp_fundo"] = None
    out["tipo_classe"] = merged["Tipo_Classe"]
    out["classe"] = merged["Classificacao"]
    out["classe_anbima"] = merged["Classificacao_Anbima"]
    out["situacao"] = merged["Situacao"]
    out["data_registro"] = merged["Data_Registro"]
    out["data_constituicao"] = merged["Data_Constituicao"]
    out["data_cancelamento"] = None
    out["data_inicio_situacao"] = merged["Data_Inicio_Situacao"]
    out["exercicio_social_inicio"] = merged["Data_Inicio_Exercicio_Social"]
    out["exercicio_social_fim"] = merged["Data_Fim_Exercicio_Social"]
    out["condominio"] = merged["Forma_Condominio"]
    out["fundo_cotas"] = None
    out["fundo_exclusivo"] = merged["Exclusivo"]
    out["publico_alvo"] = merged["Publico_Alvo"]
    out["taxa_administracao"] = None
    out["taxa_performance"] = None
    out["patrimonio_liquido"] = merged["Patrimonio_Liquido"]
    out["data_patrimonio_liquido"] = merged["Data_Patrimonio_Liquido"]
    out["administrador_nome"] = merged["Administrador"]

    out["gestor_cpf_cnpj"] = merged["CPF_CNPJ_Gestor"]

    out["controlador_nome"] = merged["Controlador"]
    out["controlador_cnpj"] = merged["CNPJ_Controlador"]

def tratar(df_fundo, df_classe, competencia):
    # Exclusao de administradores Bradesco/BEM no nivel do fundo - exclui automaticamente
    # todas as classes desse fundo quando fizermos o join.
    admin_upper = df_fundo["Administrador"].fillna("").str.upper().str.strip()
    contem_excluida = admin_upper.str.contains(
        "|".join(config.ADMINISTRADORES_EXCLUIDOS_SUBSTRINGS), regex=True, na=False
    )
    comeca_excluida = admin_upper.str.startswith(tuple(config.ADMINISTRADORES_EXCLUIDOS_PREFIXOS))
    df_fundo = df_fundo[~(contem_excluida | comeca_excluida)]

    ids_com_classe = set(df_classe["ID_Registro_Fundo"])

    df = pd.concat(
        [construir_classes(df_fundo, df_classe), construir_standalone(df_fundo, ids_com_classe)],
        ignore_index=True,
    )

    # Filtro de situacao valida (fonte usa Title Case; comparar em maiusculas).
    df = df[df["situacao"].str.upper().isin(config.SITUACOES_FUNDO_VALIDAS)]

    linhas_antes_cnpj = len(df)
    df["cnpj"] = df["cnpj"].apply(normalizar_cnpj)
    df = df[df["cnpj"].notna()]
    linhas_ignoradas = linhas_antes_cnpj - len(df)

    for coluna in ("fundo_cnpj_pai", "administrador_cnpj", "auditor_cnpj", "custodiante_cnpj", "controlador_cnpj"):
        df[coluna] = df[coluna].apply(normalizar_cnpj)

    for coluna in COLUNAS_TEXTO:
        df[coluna] = df[coluna].apply(normalizar_texto)

    for coluna in COLUNAS_DATA:
        datas = pd.to_datetime(df[coluna], errors="coerce")
        # .dt.date/.where reintroduzem NaT quando a coluna inteira e NaT - converter
        # elemento a elemento garante date ISO ou None puros.
        df[coluna] = datas.apply(lambda ts: ts.date().isoformat() if pd.notna(ts) else None)

    for coluna in COLUNAS_NUMERICAS:
        df[coluna] = pd.to_numeric(df[coluna].astype(str).str.replace(",", ".", regex=False), errors="coerce")

    for coluna in COLUNAS_BOOLEANAS:
        df[coluna] = df[coluna].map({"S": 1, "N": 0})

    df = df.drop_duplicates(subset=["cnpj"], keep="last")
    df["competencia_referencia"] = competencia
    df["ativo"] = 1

    df = df.where(pd.notnull(df), None)
    return df[COLUNAS_FUNDO], linhas_ignoradas


def persistir(conn, df):
    registros = df.to_dict(orient="records")
    cursor = conn.cursor()

    existentes = {row[0] for row in cursor.execute("SELECT cnpj FROM fundos")}
    novos_cnpjs = {r["cnpj"] for r in registros}
    inseridas = len(novos_cnpjs - existentes)
    atualizadas = len(novos_cnpjs & existentes)

    # Marca tudo como inativo antes do lote: quem continuar no universo valido e
    # reativado pelo upsert (ativo=1 ja vem em cada registro). Preserva historico
    # em vez de apagar linhas de fundos que saem do universo (cancelados etc).
    cursor.execute("UPDATE fundos SET ativo = 0 WHERE ativo = 1")

    colunas_update = [c for c in COLUNAS_FUNDO if c != "cnpj"]
    sql = f"""
        INSERT INTO fundos ({", ".join(COLUNAS_FUNDO)})
        VALUES ({", ".join(f":{c}" for c in COLUNAS_FUNDO)})
        ON CONFLICT (cnpj) DO UPDATE SET
            {", ".join(f"{c} = excluded.{c}" for c in colunas_update)},
            atualizado_em = datetime('now', 'localtime')
    """
    cursor.executemany(sql, registros)
    conn.commit()
    return inseridas, atualizadas


def atualizar_registro_fundo():
    inicio = time.monotonic()
    competencia = datetime.now().strftime("%Y%m")
    conn = obter_conexao()

    try:
        caminho_fundo, caminho_classe = baixar_e_extrair()
        df_fundo_bruto = ler_csv(caminho_fundo)
        df_classe_bruto = ler_csv(caminho_classe)
        linhas_lidas = len(df_fundo_bruto) + len(df_classe_bruto)

        df_tratado, linhas_ignoradas = tratar(df_fundo_bruto, df_classe_bruto, competencia)
        inseridas, atualizadas = persistir(conn, df_tratado)

        registrar_log(
            conn, NOME_PIPELINE, NOME_BASE_COMPETENCIA, "SUCESSO", competencia=competencia,
            linhas_lidas=linhas_lidas, linhas_processadas=len(df_tratado),
            linhas_inseridas=inseridas, linhas_atualizadas=atualizadas,
            linhas_ignoradas=linhas_ignoradas, duracao_segundos=time.monotonic() - inicio,
        )
    except Exception as exc:
        registrar_log(
            conn, NOME_PIPELINE, NOME_BASE_COMPETENCIA, "ERRO", competencia=competencia,
            mensagem_erro=str(exc), duracao_segundos=time.monotonic() - inicio,
        )
        raise
    finally:
        conn.close()


if __name__ == "__main__":
    atualizar_registro_fundo()
